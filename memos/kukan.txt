累積和
int N;
vector<int> A(N);
vector<int> SA(N+1);
SA[0] = 0;
rep(i,0,N){
    SA[i+1] = SA[i] + A[i]; //SA[i] = sum_{k=0}^{i-1} A[k];
}
int l,r;    //[l, r) = [l, r-1]の和
int sumlr = SA[r] - SA[l];  //[0,r-1] - [0, l-1] = [l,r-1]
