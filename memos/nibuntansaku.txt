bool P(int idx){
    // l: false
    // r: true
    // l < M <= r

    if(idx < M) return false;
    else if(idx >= M) return true;
}

int binary_search(){
    int left, right;    //P(left) = false, P(right) = true

    while(right -left > 1){
        int mid = left + (right - left) / 2;
        if(P(mid)) right = mid;
        else left = mid;
    }

    return right;
}
