エラトステネスのふるい(o(nlog(log n))){
  n以下の素数を全て求める

  1. 2~nまでの整数を並べる
  2. 生き残っている数の中で一番小さい(かつまだpとして使われていない)
     ものを新たにpとおき、p以外のpの倍数をすべて消していく
  3. 2の操作を繰り返していき、pがsqrt(n)を越えたら終了。最終的に生き残った者が素数

  int N;
  vector<int> is_prime(N,1); //1...prime 2...not prime
  is_prime.at(0) = 0, is_prime.at(1) = 0;
  for(int i = 2; i < N; i++){
    if(!is_prime.at(i)) continue; // 0のとき
    for(int j = i*2; j < N; j += i) is_prime.at(j) = 0; //iの倍数
  }

  #include <iostream>
#include <bitset>
using namespace std;

int main() {
    int n; cin >> n; // 入力

    // エラトステネスのふるい
    bitset<1000000> is_prime(0); // 十分大きなサイズを確保します
    for (int i = 2; i <= n; ++i) is_prime.set(i); // とりあえず全部ふるいに入れます
    for (int i = 2; i <= n; ++i) {
        if (is_prime[i]) { // 素数 i を発見したら
            for (int j = i * 2; j <= n; j += i) {
                is_prime.reset(j); // i の倍数をふるい落とす
            }
        }
    }

    // 結果出力
    for (int i = 2; i <= n; ++i) {
        if (is_prime[i]) cout << i << endl;
    }
}

}

繰り返し二乗法{
  O(log n)のx^nの計算
  x^n = \prod_{i=0} x^(2^i) として計算する．
  初期値 ans = 1
  nを2進数として見て，i桁目が1のとき ans *= x^(2^i)

  ll pow(ll x,ll n){
    ll ret = 1;
    while(n > 0){
      if(n & 1){  //nの最下位ビットが1のとき
        ret *= x;
      }
      x *= x;     //x^(2^i)を作っている
      n >>= 1;
    }
    return ret;
  }

  値が非常に大きくなるとき
  掛け算，足し算の度に剰余を取る
  #define MOD 1'000'000'007 or 998244353 //MOD以下の数どうしは，足しても32bitに収まり，かけても64bitに収まる
  MOD は素数　特に998244353はNTT-friendly な素数

  ll pow(ll x,ll n,ll mod){ //入力x,nは%mod
    ll ret = 1;
    while(n > 0){
      if(n & 1){  //nの最下位ビットが1のとき
        ret *= x % MOD;
      }
      x *= x % MOD;     //x^(2^i)を作っている
      n >>= 1;
    }
    return ret;
  }
}

mod pにおける割り算{
  逆元を利用

  フェルマーの小定理
  p:素数,a:pの倍数でない整数
  a^(p-1) == 1

  証明
  mod.pにおいてa,a*2,...,a*(p-1)をpで割った余りに1,2,...,p-1が一度ずつ現れる
  a*(a*2)*...*(a*(p-1)) == a^(p-1)*(p-1)! === (p-1)!
  (p-1)!とpは互いに素なので，両辺(p-1)!で割れ，a^(p-1) === 1 (mod p)

  a*x === b (mod p)
  をみたすxはmod.pのもとで一意にきまる
  余りが1,2,...p-1が一度ずつ現れることから従う

  mod.pにおけるaの逆元inv(a)
  a*inv(a) == 1
  フェルマーの小定理より
  inv(a) == a^(p-2)

  非再帰拡張Euclidの互除法による逆元計算
  xがaの逆元である
  a*x === 1
  a*x - 1 === 0 (mod p)
  a*x -1 == -p*y
  a*x + p*y == 1
  をみたす整数yが存在する

  ll modinv(ll a, ll m) {   //
	  ll b = m, u = 1, v = 0;
	  while (b) {
		ll t = a / b;
		a -= t * b; swap(a, b);
		u -= t * v; swap(u, v);
	  }
	  u %= m; 
	  if (u < 0) u += m;
	  return u;
  }



  mod.pにおいて
  1/a == inv(a)
  b/a == b*inv(a)
}
